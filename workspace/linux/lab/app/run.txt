# ============================================================
# imageboard 웹 애플리케이션 빌드 및 실행 명령어
# (lab 폴더의 상위 폴더에서 실행)
# ============================================================

# ------------------------------------------------------------
# 1. 빌드 (ARG 지정)
#    GIT_REPO_URL : clone 할 GitHub 저장소 주소
#    GIT_BRANCH   : clone 할 브랜치
# ------------------------------------------------------------
docker build \
  --build-arg GIT_REPO_URL=https://github.com/shared-repo/deploy.git \
  --build-arg GIT_BRANCH=master \
  -t imageboard-app:1.0 \
  ./lab/app

# ------------------------------------------------------------
# 2. 실행 (ENV 지정)
#    - imageboard-net 네트워크와 imageboard-db 컨테이너가 먼저 준비되어 있어야 함
#    - SERVER_PORT 를 변경하면 -p 옵션의 오른쪽 포트도 함께 변경
#    - SPRING_DATASOURCE_URL 은 특수문자(?, &) 때문에 따옴표로 감쌈
# ------------------------------------------------------------
docker run -d \
  --name imageboard-app \
  --network imageboard-net \
  -p 8080:8080 \
  -e SPRING_DATASOURCE_URL="jdbc:mariadb://imageboard-db:3306/imageboard?characterEncoding=UTF-8" \
  -e SPRING_DATASOURCE_USERNAME=human \
  -e SPRING_DATASOURCE_PASSWORD=human \
  -e SERVER_PORT=8080 \
  -e TZ=Asia/Seoul \
  imageboard-app:1.0
