# ============================================================
# imageboard 데이터베이스(MariaDB) 빌드 및 실행 명령어
# (lab 폴더의 상위 폴더에서 실행)
# ============================================================

# ------------------------------------------------------------
# 0. 네트워크 생성 (최초 1회)
#    - 웹 애플리케이션 컨테이너가 imageboard-db 이름으로 DB에 접속하기 위한 네트워크
# ------------------------------------------------------------
docker network create imageboard-net

# ------------------------------------------------------------
# 1. 빌드
#    - DB Dockerfile 에는 ARG 가 없으므로 --build-arg 지정 없음
#    - init.sql 이 이미지에 포함되어 최초 실행 시 자동으로 수행됨
# ------------------------------------------------------------
docker build \
  -t imageboard-db:1.0 \
  ./lab/db

# ------------------------------------------------------------
# 2. 실행 (ENV 지정)
#    MARIADB_ROOT_PASSWORD : root 계정 비밀번호
#    TZ                    : 컨테이너 시간대
#    - 컨테이너 이름은 반드시 imageboard-db (웹 애플리케이션의 접속 URL 호스트명)
#    - init.sql 은 데이터 디렉터리가 비어 있을 때(최초 실행 시) 한 번만 실행됨
# ------------------------------------------------------------
docker run -d \
  --name imageboard-db \
  --network imageboard-net \
  -p 3306:3306 \
  -e MARIADB_ROOT_PASSWORD=human \
  -e TZ=Asia/Seoul \
  imageboard-db:1.0

# ------------------------------------------------------------
# 3. 접속 확인
#    - 로컬 접속 : 컨테이너 내부에서 human 계정으로 접속
#    - 원격 접속 : 호스트에서 3306 포트로 접속 (MariaDB/MySQL 클라이언트 필요)
# ------------------------------------------------------------
docker exec -it imageboard-db mariadb -u human -phuman imageboard

mariadb -h 127.0.0.1 -P 3306 -u human -phuman imageboard
